    # 3. Apply the IAM policy bindings at the organization
      level
    for role in $FAST_ROLES; do
      gcloud organizations add-iam-policy-binding $FAST_ORG_ID
      \
        --member "$FAST_PRINCIPAL" \
        --role "$role" \
        --condition None
    done
